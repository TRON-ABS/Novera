cmake_minimum_required(VERSION 3.24)
project(novera_browser_text VERSION 1.0.1 LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Build this target with the pinned Emscripten SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG -flto")
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG -flto")
add_compile_options(-fwasm-exceptions -msimd128)
add_link_options(-fwasm-exceptions -flto)
option(NOVERA_CPU_PTHREADS "Enable private browser CPU worker pool" OFF)
if(NOVERA_CPU_PTHREADS)
  add_compile_options(-pthread)
  add_link_options(-pthread)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_WEBGPU ON CACHE BOOL "" FORCE)
set(GGML_WEBGPU_JSPI ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(NOVERA_WASM_MEMORY64 "Build the 64-bit memory runtime for large GGUFs" OFF)
set(LLAMA_WASM_MEM64 "${NOVERA_WASM_MEMORY64}" CACHE BOOL "" FORCE)
if(NOVERA_WASM_MEMORY64)
  add_compile_options("-sMEMORY64=1")
  add_link_options("-sMEMORY64=1")
  set(NOVERA_MAXIMUM_MEMORY 17179869184)
else()
  set(NOVERA_MAXIMUM_MEMORY 4294967296)
endif()
set(LLAMA_BUILD_HTML OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_COMMIT "cd959bae99ff4bed22bef2604f45241c09bbb8c8")
add_subdirectory("${NOVERA_LLAMA_SOURCE}" llama)
if(NOVERA_CPU_PTHREADS)
  target_compile_definitions(ggml-webgpu PRIVATE NOVERA_MAIN_RUNTIME_ONLY=1)
  target_compile_definitions(ggml-base PRIVATE NOVERA_MAIN_RUNTIME_ONLY=1)
  target_compile_definitions(ggml-cpu PRIVATE NOVERA_AUTO_CPU_THREADS=1)
endif()
if(NOVERA_CPU_SOURCE AND NOVERA_CPU_OPS_SOURCE)
  get_target_property(cpu_sources ggml-cpu SOURCES)
  list(REMOVE_ITEM cpu_sources ggml-cpu/ggml-cpu.c ggml-cpu/ops.cpp)
  set_property(TARGET ggml-cpu PROPERTY SOURCES ${cpu_sources} "${NOVERA_CPU_SOURCE}" "${NOVERA_CPU_OPS_SOURCE}")
  target_include_directories(ggml-cpu PRIVATE "${NOVERA_LLAMA_SOURCE}/ggml/src/ggml-cpu")
endif()
if(NOVERA_GGML_BACKEND_SOURCE)
  get_target_property(ggml_base_sources ggml-base SOURCES)
  list(REMOVE_ITEM ggml_base_sources ggml-backend.cpp)
  set_property(TARGET ggml-base PROPERTY SOURCES ${ggml_base_sources} "${NOVERA_GGML_BACKEND_SOURCE}")
  target_include_directories(ggml-base PRIVATE "${NOVERA_LLAMA_SOURCE}/ggml/src")
endif()
if(NOVERA_WEBGPU_SOURCE)
  get_target_property(webgpu_sources ggml-webgpu SOURCES)
  list(REMOVE_ITEM webgpu_sources ggml-webgpu.cpp)
  set_property(TARGET ggml-webgpu PROPERTY SOURCES ${webgpu_sources} "${NOVERA_WEBGPU_SOURCE}")
  target_include_directories(ggml-webgpu PRIVATE "${NOVERA_LLAMA_SOURCE}/ggml/src/ggml-webgpu")
  target_include_directories(ggml-webgpu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(NOVERA_WEBGPU_SHADER_INCLUDE)
  target_include_directories(ggml-webgpu BEFORE PRIVATE "${NOVERA_WEBGPU_SHADER_INCLUDE}")
endif()
add_executable(novera-text novera-text.cpp)
target_link_libraries(novera-text PRIVATE llama llama-common)
target_include_directories(novera-text PRIVATE "${NOVERA_LLAMA_SOURCE}/src")
if(NOVERA_CPU_PTHREADS)
  target_link_options(novera-text PRIVATE "-sPTHREAD_POOL_SIZE=(globalThis.noveraCpuPoolSize||0)" "-sPTHREAD_POOL_SIZE_STRICT=2")
endif()
set_target_properties(novera-text PROPERTIES SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_link_options(novera-text PRIVATE
  --no-entry -lworkerfs.js
  "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker"
  "-sALLOW_MEMORY_GROWTH=1" "-sINITIAL_MEMORY=268435456" "-sMAXIMUM_MEMORY=${NOVERA_MAXIMUM_MEMORY}"
  "-sSTACK_SIZE=8388608" "-sFILESYSTEM=1" "-sASSERTIONS=1"
  "-sEXPORTED_FUNCTIONS=['_malloc','_free','_novera_load','_novera_generate','_novera_cancel','_novera_unload','_novera_result','_novera_quantize_q4','_novera_probe_logits','_novera_probe_chunk']"
  "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap','UTF8ToString','FS','WORKERFS','HEAPU8']"
  "-sINCOMING_MODULE_JS_API=['wasmBinary','print','printErr','locateFile','onAbort']"
  "-sJSPI_EXPORTS=['novera_load','novera_generate','novera_unload','novera_probe_logits','novera_probe_chunk']")
